# services/llm/deepseek_harness.py
"""
Production DeepSeek harness with reasoning-aware tool use,
structured output, and LangSmith tracing baked in.
"""
import os, json, re
from typing import Any, Type
from langchain_core.language_models.chat_models import BaseChatModel
from langchain_core.messages import BaseMessage, HumanMessage, SystemMessage
from langchain_core.outputs import ChatResult, ChatGeneration
from langsmith import traceable
from pydantic import BaseModel
import httpx

DEEPSEEK_API_BASE = "https://api.deepseek.com/v1"
DEEPSEEK_REASONING = "deepseek-reasoner"   # R1
DEEPSEEK_CHAT      = "deepseek-chat"        # V3

class DeepSeekHarness(BaseChatModel):
    """
    DeepSeek harness implementing LangChain's BaseChatModel interface.
    Supports:
      - DeepSeek-V3 (deepseek-chat) — general
      - DeepSeek-R1 (deepseek-reasoner) — reasoning with <think>...</think>
      - Tool use (function calling)
      - JSON mode (response_format)
    """
    
    model_name: str = DEEPSEEK_CHAT
    temperature: float = 0.2
    max_tokens: int = 4096
    api_key: str = os.getenv("DEEPSEEK_API_KEY", "")
    
    class Config:
        arbitrary_types_allowed = True
    
    @property
    def _llm_type(self) -> str:
        return "deepseek"
    
    def _generate(self, messages: list[BaseMessage], **kwargs) -> ChatResult:
        payload = {
            "model": self.model_name,
            "messages": self._convert_messages(messages),
            "temperature": kwargs.get("temperature", self.temperature),
            "max_tokens": self.max_tokens,
            "stream": False,
        }
        if kwargs.get("response_format"):
            payload["response_format"] = kwargs["response_format"]
        if kwargs.get("tools"):
            payload["tools"] = kwargs["tools"]
        
        with httpx.Client(timeout=60) as client:
            r = client.post(
                f"{DEEPSEEK_API_BASE}/chat/completions",
                headers={"Authorization": f"Bearer {self.api_key}"},
                json=payload,
            )
            r.raise_for_status()
            data = r.json()
        
        choice = data["choices"][0]
        content = choice["message"]["content"]
        reasoning, final = self._split_reasoning(content)
        return ChatResult(
            generations=[ChatGeneration(
                message=HumanMessage(content=final),
                generation_info={"reasoning": reasoning, "usage": data.get("usage")},
            )]
        )
    
    @staticmethod
    def _split_reasoning(content: str) -> tuple[str, str]:
        """Split DeepSeek-R1's <think>...</think> from final answer."""
        m = re.search(r"<think>(.*?)</think>(.*)", content, re.DOTALL)
        if m:
            return m.group(1).strip(), m.group(2).strip()
        return "", content
    
    def _convert_messages(self, messages):
        out = []
        for m in messages:
            role = {"HumanMessage": "user", "AIMessage": "assistant",
                    "SystemMessage": "system"}.get(m.__class__.__name__, "user")
            out.append({"role": role, "content": m.content})
        return out


# ── Structured output via Pydantic ──────────────────────────────────
@traceable(name="deepseek.structured")
def deepseek_structured(
    harness: DeepSeekHarness,
    prompt: str,
    schema: Type[BaseModel],
    system: str = "You are a precise CBSE Class X tutor."
) -> BaseModel:
    """Forces JSON output then parses via Pydantic."""
    full = f"{system}\n\n{prompt}\n\nReturn ONLY valid JSON matching: {schema.model_json_schema()}"
    result = harness.invoke([HumanMessage(content=full)])
    raw = result.content.strip().strip("```json").strip("```")
    return schema.model_validate_json(raw)


# ── R1 reasoning helper ─────────────────────────────────────────────
@traceable(name="deepseek.reason")
def deepseek_reason(prompt: str, system: str = "") -> tuple[str, str]:
    """Returns (reasoning_chain, final_answer) from DeepSeek-R1."""
    h = DeepSeekHarness(model_name=DEEPSEEK_REASONING, temperature=0.6)
    msgs = []
    if system: msgs.append(SystemMessage(content=system))
    msgs.append(HumanMessage(content=prompt))
    result = h.invoke(msgs)
    reasoning, final = h._split_reasoning(
        result.content if not result.generation_info.get("reasoning")
        else f"<think>{result.generation_info['reasoning']}</think>{result.content}"
    )
    return reasoning, final

def make_cid(*parts: str) -> str:
    norm = "|".join(
        re.sub(r"[\s,'\"]+", "_", p.lower().strip())
        for p in parts if p
    )
    return f"cbe:{hashlib.sha256(norm.encode()).hexdigest()[:16]}"

from dataclasses import dataclass, field, asdict
from pathlib import Path
import re

@dataclass
class GDT_Token:
    text: str
    confidence: float = 1.0
    layout: dict = field(default_factory=dict)

@dataclass
class GDT_Line:
    tokens: list[GDT_Token] = field(default_factory=list)
    @property
    def text(self): return " ".join(t.text for t in self.tokens)

@dataclass
class GDT_Paragraph:
    lines: list[GDT_Line] = field(default_factory=list)
    type: str = "PARAGRAPH"
    @property
    def text(self): return "\n".join(l.text for l in self.lines)

@dataclass
class GDT_Block:
    paragraphs: list[GDT_Paragraph] = field(default_factory=list)
    type: str = "TEXT_BLOCK"

@dataclass
class GDT_Page:
    page_number: int
    blocks: list[GDT_Block] = field(default_factory=list)

@dataclass
class GDT_Document:
    doc_id: str
    source_uri: str
    pages: list[GDT_Page] = field(default_factory=list)
    detected_language: str = "en"
    
    def to_dict(self):
        return {
            "@context": "https://documentai.googleapis.com/v1",
            "@type": "Document",
            "doc_id": self.doc_id,
            "source_uri": self.source_uri,

                {"page_number": p.page_number,
                 "blocks": [{"paragraphs": [
                    {"type": par.type, "text": par.text,
                     "lines": [{"text": l.text} for l in par.lines]}

                for b in p.blocks]}
                for p in self.pages

        }

    def _parse_docai(self, pdf_path: Path, source_uri: str) -> GDT_Document:
        from google.cloud import documentai
        client = documentai.DocumentProcessorServiceClient()
        with open(pdf_path, "rb") as f:
            raw = f.read()
        result = client.process_document(
            documentai.ProcessRequest(
                name=self.processor,
                raw_document=documentai.RawDocument(content=raw, mime_type="application/pdf"),
            )
        ).document
        return self._convert_docai(result, source_uri)
    
    def _convert_docai(self, doc, source_uri: str) -> GDT_Document:
        gdt = GDT_Document(doc_id=Path(source_uri).stem, source_uri=source_uri)
        for page_idx, page in enumerate(doc.pages):
            gdt_page = GDT_Page(page_number=page_idx + 1)
            for block in page.blocks:
                gdt_block = GDT_Block()
                for para in block.paragraphs:
                    text = "".join(t.text for l in para.lines for t in t.tokens)
                    gdt_para = GDT_Paragraph(type=self._classify(text), lines=[
                        GDT_Line(tokens=[GDT_Token(text=t.text) for t in l.tokens])
                        for l in para.lines
                    ])
                    gdt_block.paragraphs.append(gdt_para)
                gdt_page.blocks.append(gdt_block)
            gdt.pages.append(gdt_page)
        return gdt
    
    def _parse_fallback(self, pdf_path: Path, source_uri: str) -> GDT_Document:
        from pypdf import PdfReader
        gdt = GDT_Document(doc_id=pdf_path.stem, source_uri=source_uri)
        reader = PdfReader(pdf_path)
        for page_idx, page in enumerate(reader.pages):
            gdt_page = GDT_Page(page_number=page_idx + 1)
            text = page.extract_text() or ""
            for raw_block in text.split("\n\n"):
                if not raw_block.strip(): continue
                gdt_block = GDT_Block()
                for raw_line in raw_block.split("\n"):
                    if not raw_line.strip(): continue
                    ptype = self._classify(raw_line)
                    para = GDT_Paragraph(type=ptype,
                        lines=[GDT_Line(tokens=[GDT_Token(text=raw_line.strip())])])
                    gdt_block.paragraphs.append(para)
                gdt_page.blocks.append(gdt_block)
            gdt.pages.append(gdt_page)
        return gdt
    
    @staticmethod
    def _classify(text: str) -> str:
        t = text.strip()
        if re.match(r"^(\d+(\.\d+)?)\.?\s+[A-Z]", t): return "HEADING"
        if re.match(r"^[\(\[][a-z\d]+[\)\]]", t):        return "LIST_ITEM"
        if re.match(r"^(Theorem|Definition|Note|Proof)[:\s]", t, re.I): return "DEFINITION"
        if re.search(r"[=+\-×÷√π∑∫]", t) and ("$" in t or "\\" in t or "^" in t):
            return "FORMULA"
        return "PARAGRAPH"

# services/graph_engineering/graphify.py
"""
Graphify: walks a Google Document Tree, extracts OKF v0.2 entities,
builds relations, normalizes text (air-fillers removed), and
correlates every chunk to the CBSE Class X syllabus.
"""
import re
from typing import Optional
from langsmith import traceable
from okf_schema import (OKFEntity, OKFRelation, OKFGraph, CBSEEntityType,
                        RelationType, make_cid)
from gdt_parser import GDT_Document, GDT_Paragraph

# ── Normalizer (air-fillers + dedup prep) ─────────────────────────
AIR_FILLER_RE = re.compile(
    r"\s+|"
    r"(?<=\w),(?=\w)|"
    r"[\u2018\u2019\u201C\u201D'\"]"
)
def normalize(text: str) -> str:
    text = AIR_FILLER_RE.sub(" ", text).strip().lower()
    return re.sub(r"\s+", " ", text)

HEADING_RE = re.compile(r"^(\d+)\.(\d+)(?:\.(\d+))?\s+(.+)$")

class GraphifyEngine:
    def __init__(self, subject: str, class_no: int = 10,
                 syllabus_chapter_names: list[str] = None):
        self.subject = subject
        self.class_no = class_no
        self.entities: dict[str, OKFEntity] = {}
        self.relations: list[OKFRelation] = []
        self.subject_root = make_cid("subject", f"class{class_no}", subject)
        self.chapter_ids: dict[str, str] = {}
        self.syllabus_chapter_names = syllabus_chapter_names or []
    
    @traceable(name="graphify.run")
    def graphify(self, doc: GDT_Document) -> OKFGraph:
        self._add(OKFEntity(**{
            "@id": self.subject_root, "@type": CBSEEntityType.SUBJECT_NODE,
            "name": f"CBSE Class {self.class_no} {self.subject.title()}",
            "alternateName": [self.subject.title()],
            "inLanguage": "en-IN",
        }))
        
        current_chapter_id: Optional[str] = None
        current_topic_id: Optional[str] = None
        
        for page in doc.pages:
            for block in page.blocks:
                for para in block.paragraphs:
                    if not para.text.strip(): continue
                    txt = normalize(para.text)
                    
                    if para.type == "HEADING":
                        current_chapter_id, current_topic_id = \
                            self._handle_heading(txt, page.page_number)
                    elif para.type == "DEFINITION":
                        self._handle_definition(txt, current_topic_id, page.page_number)
                    elif para.type == "FORMULA":
                        self._handle_formula(txt, current_topic_id, page.page_number)
                    elif para.type in ("PARAGRAPH", "LIST_ITEM"):
                        self._handle_chunk(txt, current_topic_id, page.page_number)
        
        return OKFGraph(
            graph_id=self.subject_root,
            class_level=self.class_no,
            subject=self.subject,
            entities=list(self.entities.values()),
            relations=self.relations,
            source_document={"url": doc.source_uri, "doc_id": doc.doc_id,
                              "page_count": len(doc.pages)},
        )
    
    def _handle_heading(self, txt: str, page: int):
        m = HEADING_RE.match(txt)
        if not m: return None, None
        major, minor, sub, title_raw = m.groups()
        title = title_raw.strip()
        
        if minor is None:
            chapter_id = make_cid("chapter", self.subject_root, f"ch{major}", title)
            official = self._match_syllabus(title)
            self._add(OKFEntity(**{
                "@id": chapter_id, "@type": CBSEEntityType.CHAPTER,
                "name": official or title.title(),
                "alternateName": [title] if official else [],
                "sourceCitation": {"page": page, "number": major},
            }))
            self._rel(self.subject_root, RelationType.HAS_CHAPTER, chapter_id)
            self.chapter_ids[major] = chapter_id
            return chapter_id, None
        
        parent_ch = self.chapter_ids.get(major)
        if not parent_ch: return None, None
        topic_id = make_cid("topic", parent_ch, title)
        self._add(OKFEntity(**{
            "@id": topic_id, "@type": CBSEEntityType.TOPIC,
            "name": title.title(),
            "sourceCitation": {"page": page},
        }))
        self._rel(parent_ch, RelationType.HAS_TOPIC, topic_id)
        return parent_ch, topic_id
    
    def _handle_definition(self, txt: str, topic_id: Optional[str], page: int):
        m = re.match(r"(theorem|definition|note|proof)[:\s]+(.+)", txt, re.I)
        if not m: return
        kind, body = m.groups()
        etype = {"theorem": CBSEEntityType.THEOREM,
                 "definition": CBSEEntityType.DEFINITION,
                 "note": CBSEEntityType.CONCEPT,
                 "proof": CBSEEntityType.CONCEPT}.get(kind.lower())
        if not etype: return
        cid = make_cid(etype.value.split(":")[1], body)
        self._add(OKFEntity(**{
            "@id": cid, "@type": etype,
            "name": body[:120].title(), "description": body,
            "sourceCitation": {"page": page},
        }))
        if topic_id: self._rel(topic_id, RelationType.EXPLAINS, cid)
    
    def _handle_formula(self, txt: str, topic_id: Optional[str], page: int):
        cid = make_cid("formula", txt)
        self._add(OKFEntity(**{
            "@id": cid, "@type": CBSEEntityType.FORMULA,
            "name": txt[:80], "description": txt,
            "sourceCitation": {"page": page},
        }))
        if topic_id: self._rel(topic_id, RelationType.EXPLAINS, cid)
    
    def _handle_chunk(self, txt: str, topic_id: Optional[str], page: int):
        cid = make_cid("chunk", txt)
        self._add(OKFEntity(**{
            "@id": cid, "@type": CBSEEntityType.CONCEPT,
            "name": txt[:80], "description": txt,
            "sourceCitation": {"page": page},
        }))
        if topic_id:
            self._rel(topic_id, RelationType.EXPLAINS, cid)
    
    def _add(self, entity: OKFEntity):
        """Canonical @id dedup. Same normalized text → same @id."""
        if entity.id not in self.entities:
            self.entities[entity.id] = entity
        else:
            existing = self.entities[entity.id]
            for alt in [entity.name] + entity.alternateName:
                if alt and alt not in existing.alternateName and alt != existing.name:
                    existing.alternateName.append(alt)
    
    def _rel(self, subj: str, pred: RelationType, obj: str):
        if not any(r.subject == subj and r.predicate == pred and r.object == obj
                   for r in self.relations):
            self.relations.append(OKFRelation(**{"@id": subj, "predicate": pred, "object": obj}))
    
    def _match_syllabus(self, parsed_title: str) -> Optional[str]:
        from difflib import get_close_matches
        m = get_close_matches(parsed_title, self.syllabus_chapter_names, n=1, cutoff=0.7)
        return m[0] if m else None

Two-stage dedup. Hard Class X boundary.

Hash-based dedup via canonical @id, then embedding-based near-dedup for synonyms. Every chunk then gets a cbe:alignedTo relation back to the CBSE syllabus node.

Semantic Deduplicator

import numpy as np
from langsmith import traceable
from okf_schema import OKFGraph, CBSEEntityType

class SemanticDeduplicator:
    """Two-stage: (1) canonical @id exact dedup (2) embedding near-dedup."""
    def __init__(self, embed_fn, threshold: float = 0.92):
        self.embed_fn = embed_fn
        self.threshold = threshold
    
    @traceable(name="dedup.semantic")
    def dedup(self, graph: OKFGraph) -> OKFGraph:
        for etype in CBSEEntityType:
            ents = [e for e in graph.entities if e.type == etype]
            if len(ents) < 2: continue
            texts = [f"{e.name} {e.description or ''}" for e in ents]
            embs = np.array([self.embed_fn(t) for t in texts])
            norms = np.linalg.norm(embs, axis=1, keepdims=True)
            embs = embs / (norms + 1e-9)
            sims = embs @ embs.T
            
            merge_map = {}
            for i in range(len(ents)):
                if ents[i].id in merge_map: continue
                for j in range(i + 1, len(ents)):
                    if sims[i][j] > self.threshold:
                        merge_map[ents[j].id] = ents[i].id
                        for alt in [ents[j].name] + ents[j].alternateName:
                            if alt not in ents[i].alternateName and alt != ents[i].name:
                                ents[i].alternateName.append(alt)
            
            for r in graph.relations:
                if r.subject in merge_map: r.subject = merge_map[r.subject]
                if r.object in merge_map: r.object = merge_map[r.object]
            graph.entities = [e for e in graph.entities if e.id not in merge_map]
        return graph

            target = self.index.get(normalize(e.name))
            if not target:

                if m: target = self.index[m[0]]
            if target:
                graph.relations.append({
                    "@id": e.id, "predicate": RelationType.ALIGNED_TO,
                    "object": target, "confidence": 0.9,
                })
        return graph

        scores = {}
        for i, h in enumerate(vector_hits):
            scores[h.id] = scores.get(h.id, 0) + 1 / (k + i + 1)
        for i, g in enumerate(graph_hits):
            scores[g["id"]] = scores.get(g["id"], 0) + 1 / (k + i + 1)
        return sorted(scores.items(), key=lambda x: -x[1])

# services/streaks/streak_service.py
from datetime import date, timedelta
from langsmith import traceable
import asyncpg

class StreakService:
    def __init__(self):
        self.pool = None
    
    @traceable(name="streak.record_activity")
    async def record_activity(self, student_id: str, activity_type: str):
        today = date.today()
        async with self.pool.acquire() as conn:
            row = await conn.fetchrow(
                "SELECT current_streak, longest_streak, last_active_date FROM streaks WHERE student_id=$1",
                student_id)
            
            if not row:
                await conn.execute(
                    "INSERT INTO streaks (student_id, current_streak, longest_streak, last_active_date, total_xp) VALUES ($1, 1, 1, $2, 10)",
                    student_id, today)
                return {"current": 1, "xp": 10}
            
            last = row['last_active_date']
            current = row['current_streak']
            
            if last == today:
                xp = self._xp_for_activity(activity_type)
                await conn.execute("UPDATE streaks SET total_xp = total_xp + $2 WHERE student_id=$1",
                                       student_id, xp)
                return {"current": current, "xp": xp, "already_active_today": True}
            
            if last == today - timedelta(days=1):
                new_streak = current + 1
            elif last < today - timedelta(days=1):
                new_streak = 1
            else:
                new_streak = current
            
            xp = self._xp_for_activity(activity_type) + (10 if new_streak > current else 0)
            longest = max(row['longest_streak'], new_streak)
            
            await conn.execute("""UPDATE streaks SET current_streak=$2, longest_streak=$3,
                    last_active_date=$4, total_xp=total_xp+$5 WHERE student_id=$1""",
                student_id, new_streak, longest, today, xp)
            
            await self._check_badges(conn, student_id, new_streak, xp)
            return {"current": new_streak, "longest": longest, "xp_earned": xp}
    
    def _xp_for_activity(self, activity: str) -> int:
        return {"theorem_explain": 15, "problem_solve": 25,
                "experiment_story": 20, "history_scene": 20,
                "social_episode": 20, "music_pair": 10,
                "complete_quiz": 50, "perfect_score": 100}.get(activity, 10)
    
    async def _check_badges(self, conn, student_id: str, streak: int, xp: int):
        if streak >= 7: await self._award_badge(conn, student_id, "week_warrior")
        if streak >= 30: await self._award_badge(conn, student_id, "month_master")
        if streak >= 365: await self._award_badge(conn, student_id, "year_legend")
        if xp >= 1000: await self._award_badge(conn, student_id, "xp_1000")
    
    async def _award_badge(self, conn, student_id: str, badge_code: str):
        await conn.execute("""INSERT INTO student_badges (student_id, badge_code)
            VALUES ($1, $2) ON CONFLICT DO NOTHING""", student_id, badge_code)

18 · DESIGN PHILOSOPHY

"Education should feel like planting a seed, watering it daily, and watching a flower bloom."

We reject the spreadsheet paradigm. We reject the leaderboard-treadmill paradigm. We reject the dopamine-machine paradigm. What we embrace is The Garden Paradigm: every student is a garden. Every concept they learn is a seed. Every day they show up is sunlight.

The Garden Paradigm
Students don't fill buckets — they grow gardens. Streaks aren't a number; they are consecutive days of growth. Mastery isn't a percentage; it is a flower in full bloom. The three stakeholders experience this same world through three different emotional lenses.

23 · ACCESSIBILITY & INCLUSION

Built for every student. Every family. Every device.

Inclusive design is not a feature — it's the foundation.

👁️ Visual Accessibility

  • Color contrast: WCAG 2.2 AA (4.5:1 text); AAA (7:1) body text where possible.
  • Color blindness: All state indicators use shape + color + label. Red flags have ⚠️ icon.
  • Font scaling: All UI respects browser font size. Layout reflows gracefully up to 200% zoom.
  • Dark mode: Full dark mode with warm dark blues (not pure black).

🧠 Cognitive Accessibility

  • Reading level: Microcopy below grade 8 reading level.
  • No time pressure: Quizzes not timed by default. Optional timer for challenge.
  • Plain language: Avoid jargon. "Continue" not "Resume progression."
  • Predictable: Same navigation across all subjects.

✋ Motor Accessibility

  • Touch targets: Minimum 44×44px / 48×48dp.
  • Keyboard navigation: Full keyboard support. Visible focus rings (2px chakra-blue).
  • Voice control: Compatible with iOS Voice Control and Android Voice Access.

🔊 Screen Reader & Multilingual

  • ARIA: All icons have aria-label. Live regions announce streak changes.
  • Math: MathML rendering with fallback text description.
  • Hindi: Full UI translation. Devanagari typography tested.
  • Future: Roadmap for Tamil, Telugu, Bengali, Marathi.

from fastapi import FastAPI, BackgroundTasks
from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel

from services.streaks.streak_service import StreakService

app = FastAPI(title="CBSE Class X (AY 2026-2027) Platform")

streaks = StreakService()

class AskReq(BaseModel):
    question: str
    user_id: str
    subject: Optional[str] = None
    chapter_no: Optional[int] = None
    intent: Optional[str] = None
    theorem_name: Optional[str] = None
    experiment_id: Optional[str] = None
    event_id: Optional[str] = None
    concept_id: Optional[str] = None
    lesson_id: Optional[str] = None

class AskRes(BaseModel):
    answer_html: str
    formatted_html: str
    intent_used: str
    xp_earned: int
    streak: dict
    badge_unlocked: Optional[str] = None

@app.post("/ask", response_model=AskRes)
async def ask(req: AskReq, bg: BackgroundTasks):
    state = {
        "messages": [{"role": "user", "content": req.question}],
        "user_id": req.user_id, "student_class": 10,
        "subject": req.subject, "chapter_no": req.chapter_no,
        "intent": req.intent, "theorem_name": req.theorem_name,
        "experiment_id": req.experiment_id, "event_id": req.event_id,
        "concept_id": req.concept_id, "lesson_id": req.lesson_id,
        "vector_hits": [], "graph_hops": [],
        "attempt": 0, "max_attempts": 3,
    }
    result = await orchestrator.graph.ainvoke(
        state, config={"configurable": {"thread_id": req.user_id}})
    return AskRes(
        answer_html=result.get("answer", {}).get("story", str(result.get("answer", ""))),
        formatted_html=result.get("formatted_html", ""),
        intent_used=result.get("intent", ""),
        xp_earned=result.get("xp_earned", 0),
        streak=result.get("streak", {}),
        badge_unlocked=result.get("badge_unlocked"),
    )

export default {
  async fetch(req, env) {
    const cache = caches.default;
    const hit = await cache.match(req);
    if (hit) return hit;
    const res = await fetch(`${env.ORIGIN}${new URL(req.url).pathname}`);
    const cached = new Response(res.body, res);

# scripts/evaluate_langsmith.py
from langsmith import evaluate
from langsmith.evaluation import LangChainStringEvaluator
from services.observability.langsmith_setup import client
from agents.rag_agent import RAGAgent

def load_dataset():
    return client().read_dataset(dataset_name="cbse-class-x-eval")

def predict(inputs: dict) -> dict:
    agent = RAGAgent()
    result = agent.graph.invoke({
        "messages": [{"role": "user", "content": inputs["question"]}],
        "student_class": 10, "subject": inputs.get("subject"),
        "vector_hits": [], "graph_hops": [],
        "attempt": 0, "max_attempts": 3,
    })
    return {"answer": result["formatted_html"], "citations": result["vector_hits"]}

def in_syllabus(run, example):
    return {"score": 1.0 if run.outputs.get("guard_passed") else 0.0}

results = evaluate(
    predict, data=load_dataset(),
    evaluators=[LangChainStringEvaluator("faithfulness"),
                LangChainStringEvaluator("answer_relevancy"),
                LangChainStringEvaluator("context_precision"),
                in_syllabus],
    experiment_prefix="cbse-x-v4",
)
print(results)
